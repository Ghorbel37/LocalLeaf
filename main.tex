%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% main.tex: entry point of the report
%
%   main.tex          <- this file: packages, settings, document outline
%   metadata.tex      <- title, author, date... (edit this first)
%   chapters/         <- one file per chapter
%   images/           <- figures
%   references.bib    <- bibliography
%
% Build: Ctrl+Alt+B in VS Code, or "latexmk" in a terminal.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\documentclass[12pt, a4paper, oneside]{report}

% --- Language and fonts --------------------------------------------------
\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage[english]{babel}     % use [french] for a French report
\usepackage{lmodern}
\usepackage{microtype}          % nicer spacing and margins

% --- Page layout ---------------------------------------------------------
\usepackage[margin=2.5cm]{geometry}
\usepackage{setspace}
\onehalfspacing
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\nouppercase{\leftmark}}
\fancyfoot[C]{\thepage}
\renewcommand{\headrulewidth}{0.4pt}
\fancypagestyle{plain}{\fancyhf{}\fancyfoot[C]{\thepage}\renewcommand{\headrulewidth}{0pt}}

% --- Figures and tables --------------------------------------------------
\usepackage{graphicx}
\graphicspath{{images/}}
\usepackage{float}              % [H] placement
\usepackage{caption}
\usepackage{subcaption}
\usepackage{wrapfig}            % text flowing around a figure
\usepackage{booktabs}           % professional table rules
\usepackage{tabularx}           % tables as wide as the text
\usepackage{multirow}           % cells spanning several rows
\usepackage{longtable}          % tables spanning several pages
\usepackage[table]{xcolor}      % colors, including table rows
\usepackage{tikz}               % diagrams
\usetikzlibrary{positioning, arrows.meta, shapes.geometric}
\usepackage{pgfplots}           % charts
\pgfplotsset{compat=1.18}

% --- Boxes, lists, quotes ------------------------------------------------
\usepackage[most]{tcolorbox}    % colored boxes
\newtcolorbox{notebox}[1][Note]{colback=blue!5, colframe=blue!60!black, fonttitle=\bfseries, title=#1}
\newtcolorbox{warningbox}[1][Warning]{colback=orange!8, colframe=orange!80!black, fonttitle=\bfseries, title=#1}
\usepackage{enumitem}           % customizable lists
\usepackage{csquotes}           % \enquote{...} and display quotes

% --- Math ----------------------------------------------------------------
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{amsthm}
\newtheorem{theorem}{Theorem}[chapter]
\theoremstyle{definition}
\newtheorem{definition}[theorem]{Definition}
\newtheorem{example}[theorem]{Example}

% --- Code highlighting (needs Pygments and -shell-escape) ----------------
\usepackage{minted}
\setminted{frame=lines, framesep=2mm, fontsize=\small, linenos, breaklines}
\renewcommand{\listingscaption}{Listing}

% --- Acronyms ------------------------------------------------------------
\usepackage{acro}
\input{acronyms}

% --- Bibliography --------------------------------------------------------
\usepackage[sorting=none, backend=biber]{biblatex}
\addbibresource{references.bib}

% --- Links (load last) ---------------------------------------------------
\usepackage[hidelinks]{hyperref}
\usepackage[nameinlink]{cleveref}
\crefname{listing}{listing}{listings}
\Crefname{listing}{Listing}{Listings}
\crefname{definition}{definition}{definitions}
\crefname{example}{example}{examples}

% --- Report information --------------------------------------------------
\input{metadata}

\hypersetup{
  pdftitle  = {\reportTitle},
  pdfauthor = {\reportAuthor},
}

% =========================================================================
\begin{document}

% --- Front matter (roman page numbers) -----------------------------------
\pagenumbering{roman}
\input{chapters/00-title-page}
\input{chapters/01-abstract}

\tableofcontents
\listoffigures
\listoftables
\listoflistings
\printacronyms[heading=chapter*, name=List of Acronyms]

% --- Main matter (arabic page numbers) -----------------------------------
\cleardoublepage
\pagenumbering{arabic}

\input{chapters/02-introduction}
\input{chapters/03-figures}
\input{chapters/04-tables}
\input{chapters/05-code-and-math}
\input{chapters/06-writing-tools}
\input{chapters/07-conclusion}

% --- Back matter ---------------------------------------------------------
\printbibliography[heading=bibintoc]

\appendix
\input{chapters/08-appendix}

\end{document}
