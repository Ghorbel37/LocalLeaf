%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% main.tex: entry point of the report
%
%   main.tex          <- this file: packages, settings, document outline
%   metadata.tex      <- title, author, date... (edit this first)
%   chapters/         <- one file per chapter
%   images/           <- figures
%   references.bib    <- bibliography
%
% Build: Ctrl+Alt+B in VS Code, or "latexmk" in a terminal.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\documentclass[12pt, a4paper, oneside]{report}

% --- Language and fonts --------------------------------------------------
\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage[english]{babel}     % use [french] for a French report
\usepackage{lmodern}
\usepackage{microtype}          % nicer spacing and margins

% --- Page layout ---------------------------------------------------------
\usepackage[margin=2.5cm]{geometry}
\usepackage{setspace}
\onehalfspacing
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\nouppercase{\leftmark}}
\fancyfoot[C]{\thepage}
\renewcommand{\headrulewidth}{0.4pt}
\fancypagestyle{plain}{\fancyhf{}\fancyfoot[C]{\thepage}\renewcommand{\headrulewidth}{0pt}}

% --- Figures and tables --------------------------------------------------
\usepackage{graphicx}
\graphicspath{{images/}}
\usepackage{float}              % [H] placement
\usepackage{caption}
\usepackage{subcaption}
\usepackage{booktabs}           % professional table rules
\usepackage{tabularx}
\usepackage{xcolor}
\usepackage{tikz}

% --- Math ----------------------------------------------------------------
\usepackage{amsmath}
\usepackage{amssymb}

% --- Code highlighting (needs Pygments and -shell-escape) ----------------
\usepackage{minted}
\setminted{frame=lines, framesep=2mm, fontsize=\small, linenos, breaklines}

% --- Acronyms ------------------------------------------------------------
\usepackage{acro}
\input{acronyms}

% --- Bibliography --------------------------------------------------------
\usepackage[sorting=none, backend=biber]{biblatex}
\addbibresource{references.bib}

% --- Links (load last) ---------------------------------------------------
\usepackage[hidelinks]{hyperref}
\usepackage[nameinlink]{cleveref}
\crefname{listing}{listing}{listings}
\Crefname{listing}{Listing}{Listings}

% --- Report information --------------------------------------------------
\input{metadata}

\hypersetup{
  pdftitle  = {\reportTitle},
  pdfauthor = {\reportAuthor},
}

% =========================================================================
\begin{document}

% --- Front matter (roman page numbers) -----------------------------------
\pagenumbering{roman}
\input{chapters/00-title-page}
\input{chapters/01-abstract}

\tableofcontents
\listoffigures
\listoftables
\printacronyms[heading=chapter*, name=List of Acronyms]

% --- Main matter (arabic page numbers) -----------------------------------
\cleardoublepage
\pagenumbering{arabic}

\input{chapters/02-introduction}
\input{chapters/03-example-chapter}
\input{chapters/04-conclusion}

% --- Back matter ---------------------------------------------------------
\printbibliography[heading=bibintoc]

\appendix
\input{chapters/05-appendix}

\end{document}
